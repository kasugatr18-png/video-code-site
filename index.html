    <h2>コードを入力して動画を見る</h2>

    <input type="text" id="code-input" placeholder="20桁のコードを入力" maxlength="20">

    <button onclick="verifyCode()">送信</button>

    <div id="error-msg"></div>

</div>


<script>

    // Google Apps ScriptのウェブアプリURL

    const GAS_URL = "YOUR_GAS_WEB_APP_URL";


    async function verifyCode() {

        const codeInput = document.getElementById("code-input").value.trim();

        const errorMsg = document.getElementById("error-msg");

        errorMsg.textContent = "";


        // 形式判定（英数字混ざった20桁かチェック）

        const regex = /^(?=.*[0-9])(?=.*[A-Za-z])[A-Za-z0-9]{20}$/;

        if (!regex.test(codeInput)) {

            errorMsg.textContent = "コードは20桁で、数字と英語（大文字・小文字）の両方を必ず含めてください。";

            return;

        }


        errorMsg.style.color = "blue";

        errorMsg.textContent = "確認中...";


        try {

            // GASへ問い合わせ

            const response = await fetch(GAS_URL, {

                method: "POST",

                body: JSON.stringify({ code: codeInput })

            });

            const result = await response.json();


            if (result.success) {

                alert("コードが確認されました！動画ページへ移動します。");

                window.location.href = result.url; // 返ってきた動画URLへジャンプ

            } else {

                errorMsg.style.color = "red";

                errorMsg.textContent = result.message;

            }

        } catch (err) {

            console.error(err);

            errorMsg.style.color = "red";

            errorMsg.textContent = "通信エラーが発生しました。もう一度お試しください。";

        }

    }

</script>
